מָה הַשִּׂמְחָה?
בְּסִיּוּם חַג הַסּוּכּוֹת אֲנַחְנוּ מְסַיְּימִים אֶת הַתּוֹרָה. הַחֵל מִשְּׁלֹהִי זְמַן הַגְּאוֹנִים אֲנַחְנוּ חוֹגְגִים אֶת שְׁמִינִי עֲצֶרֶת כְּ"שִׂמְחַת תּוֹרָה". בְּנֵי אֶרֶץ יִשְׂרָאֵל לֹא יָכְלוּ לַחְגֹּג שִׂמְחַת תּוֹרָה, מִשּׁוּם שֶׁהָיוּ מְסַיְּימִים אֶת קְרִיאַת הַתּוֹרָה רַק פַּעַם בְּשָׁלוֹשׁ שָׁנִים. אֲבָל לְפִי מִנְהַג בָּבֶל, שֶׁהִתְקַבֵּל בְּהֶמְשֵׁךְ בְּכָל תְּפוּצוֹת יִשְׂרָאֵל, קוֹרְאִים בְּכָל שָׁבוּעַ פָּרָשָׁה וּמְסַיְּימִים אֶת הַתּוֹרָה בְּשִׂמְחַת תּוֹרָה.
שְׁמִינִי עֲצֶרֶת אֵינוֹ חֵלֶק מֵחַג הַסּוּכּוֹת. אֵין יוֹשְׁבִים בּוֹ בַּסוּכָּה, וְאֵין נוֹטְלִים אֶת אַרְבַּעַת הַמִּינִים. בְּחָג הַזֶּה אֲנַחְנוּ נֶעֱצָרִים לִפְנֵי ה' וּשְׂמֵחִים לְפָנָיו. הַקֶּשֶׁר לְסִיּוּם הַתּוֹרָה וְסִיּוּם סֵפֶר דְּבָרִים לֹא הִתְחִיל מִימֵי הַגְּאוֹנִים. הוּא מוֹפִיעַ כְּבָר בַּתּוֹרָה עַצְמָהּ: מִצְוַת הַקְהֵל, הַמִּצְווָה הַיְּחִידָה מִן הַתּוֹרָה הַמְּחַיֶּיבֶת "קְרִיאַת הַתּוֹרָה" צִבּוּרִית (קְרִיאַת הַתּוֹרָה לְאֹרֶךְ הַשָּׁנָה הִיא תַּקָּנַת מֹשֶׁה רַבֵּנוּ, וּבְהֶמְשֵׁךְ תַּקָּנַת עֶזְרָא), שֶׁבָּהּ קוֹרֵא הַמֶּלֶךְ אֶת סֵפֶר מִשְׁנֵה תוֹרָה בְּחַג הַסּוּכּוֹת.
מַדּוּעַ בְּסוּכּוֹת? וּמַדּוּעַ סֵפֶר מִשְׁנֵה תוֹרָה? חַכְמֵי הַקַּבָּלָה הִגְדִּירוּ אֶת הַסוּכָּה כְּ"צִלָּא דִּמְהֵימְנוּתָא". אַךְ מַדּוּעַ הַסוּכָּה נֶחְשֶׁבֶת לְצֵל הָאֱמוּנָה? אוֹר הַמַּקִּיף שֶׁל הַשְּׁכִינָה?
הַהֶסְבֵּר הוּא, שֶׁמִּצְוַת הַשִּׂמְחָה בֶּחָג הִיא "לִשְׂמֹחַ לִפְנֵי ה' שִׁבְעַת יָמִים", וּמִן הַצַּד הַשֵּׁנִי: "בַּסוּכּוֹת תֵּשְׁבוּ שִׁבְעַת יָמִים". וְכָאן מִתְעוֹרֶרֶת שְׁאֵלָה: אֵיךְ אֶפְשָׁר לְקַיֵּים אֶת שְׁתֵּי הַמִּצְווֹת הָאֵלּוּ יַחְדָּיו? הַשִּׂמְחָה לִפְנֵי ה' הִיא בַּמִּקְדָּשׁ, אֲבָל צְרִיכִים לָשֶׁבֶת בַּסוּכּוֹת שִׁבְעַת יָמִים?
הַתְּשׁוּבָה הִיא, שֶׁחֵלֶק מֵעִנְיַן מִצְוַת הַסוּכָּה הוּא שִׁחְזוּר שֶׁל דּוֹר הַמִּדְבָּר, "כִּי בַסֻּכּוֹת הוֹשַׁבְתִּי אֶת בְּנֵי יִשְׂרָאֵל", "עוֹד אוֹשִׁיבְךָ בָאֹהָלִים כִּימֵי מוֹעֵד". כַּאֲשֶׁר אָדָם רוֹאֶה בָּתִּים עַל גִּבְעָה, לְדֻגְמָא, הוּא מֵבִין שֶׁיֵּשׁ כָּאן כְּפָר, אֲנָשִׁים גָּרִים כָּאן. אֲבָל כַּאֲשֶׁר הוּא רוֹאֶה אֹהָלִים וְסוּכּוֹת פְּזוּרִים עַל פְּנֵי הַשָּׂדֶה, הוּא מֵבִין שֶׁיֵּשׁ כָּאן אֵיזוֹ שֶׁהִיא עֲבוֹדָה, שֶׁאַנְשֵׁי מָקוֹם מְסֻיָּם, אוֹ פּוֹעֲלִים שֶׁל אָדוֹן מְסֻיָּם, מִתְקַבְּצִים כָּאן לְשֵׁם מְלֶאכֶת הָאָסִיף, אוֹ לְשֵׁם אֵיזוֹ חֲגִיגָה.
הַסוּכּוֹת הֵן דִּירוֹת עֲרַאי, שֶׁבְּהֶכְרֵחַ טְפֵלוֹת לְאֵיזֶשֶׁהוּ מָקוֹם קֶבַע. כַּאֲשֶׁר בְּנֵי יִשְׂרָאֵל עָלוּ לָרֶגֶל, הֵם כְּמוֹ צֹאן, כְּצֹאן יְרוּשָׁלַיִם בְּמוֹעֲדֶיהָ. אַף עַל פִּי שֶׁהֵם מִתְגּוֹרְרִים בְּסוּכּוֹת וּבְאֹהָלִים סְבִיב הָעִיר, כָּל הַסוּכּוֹת הָאֵלּוּ טְפֵלוֹת לַמִּקְדָּשׁ, שָׁם שׁוֹכֶנֶת הַשְּׁכִינָה. וְלָכֵן כְּשֶׁיּוֹצְאִים מֵהַבַּיִת לַסוּכָּה, נֶהְפָּכִים לְצֹאנוֹ שֶׁל ה'. הַסוּכָּה טְפֵלָה לִמְקוֹם הַמִּקְדָּשׁ, הִיא צִלָּא דִּמְהֵימְנוּתָא. הִיא מְקוֹם מְנוּחָתָם וַאֲכִילָתָם שֶׁל הַמִּתְקַבְּצִים מִכָּל הָאָרֶץ כְּדֵי לִשְׁכֹּן בַּמִּקְדָּשׁ וְלִרְאוֹת פְּנֵי שְׁכִינָה.
וְכָךְ מוֹצְאִים אָנוּ בְּעוֹלֵי בָּבֶל, שֶׁלֹּא יָדְעוּ שֶׁיּוֹשְׁבִים בְּסוּכּוֹת. הַסִּבָּה הִיא, מִשּׁוּם שֶׁהַמִּצְווָה נִתְפְּסָה אֵצֶל בְּנֵי יִשְׂרָאֵל רַק כְּטָפֵלוּת לַמִּקְדָּשׁ וְלַשִּׂמְחָה לִפְנֵי ה', וּכְשֶׁחָרַב הַבַּיִת, חָשְׁבוּ שֶׁאֵין טַעַם בַּסוּכָּה, כִּי אֵין לְמָה לְהִטַּפֵּל. זוֹ כַּמּוּבָן טָעוּת מִבְּחִינָה הִלְכָתִית, וְלָכֵן הַדְּגֵשׁ הוּא עַל "מָצְאוּ כָתוּב": בְּאוֹתוֹ דּוֹר הוּבַן שֶׁאֵין לְהִסְתַּמֵּךְ רַק עַל מַה שֶּׁרְגִילִים וּמְקֻבָּל, אֶלָּא שֶׁהַתּוֹרָה שֶׁבְּעַל פֶּה מִתְיַשֶּׁבֶת עִם הַתּוֹרָה שֶׁבִּכְתָב, וְכָל מַה שֶּׁכָּתוּב חַיָּיב לְהִתְקַיֵּים. מִתּוֹךְ הֲבָנָה זוֹ, מוּבָן הַקֶּשֶׁר לְסֵפֶר מִשְׁנֵה תוֹרָה. אִם בְּסוּכּוֹת בְּנֵי יִשְׂרָאֵל שָׁבִים וּמְשַׁחְזְרִים אֶת הֱיוֹתָם צֹאנוֹ שֶׁל ה' בַּמִּדְבָּר, שֶׁטְּפֵלִים לַמִּשְׁכָּן, וּכְאִילּוּ אֵין לָהֶם עַכְשָׁו לֹא שָׂדוֹת וְלֹא כְּרָמִים, לֹא בָּתִּים וְלֹא עָרִים, אֶלָּא הֵם יוֹשְׁבִים בְּאֹהָלִים וּבְדִירוֹת עֲרַאי סְבִיב הַמִּשְׁכָּן וּבְכָל מָקוֹם שֶׁהֵם, הִינָם שְׂמֵחִים לִפְנֵי ה' - הֲרֵי זֶה הַזְּמַן לְהַאֲזִין לִנְאוּמָיו שֶׁל הָרוֹעֶה הַגָּדוֹל, מֹשֶׁה בֶּן עַמְרָם. וְלָכֵן זֶה הַזְּמַן הַמַּתְאִים לְשִׂמְחַת תּוֹרָה שֶׁבִּכְתָב בִּכְלָל, שֶׁבְּסוּכּוֹת אָנוּ חוֹזְרִים לִהְיוֹת כְּמוֹ בְּנֵי יִשְׂרָאֵל דּוֹר מְקַבְּלֵי הַתּוֹרָה.
וּבְכָל מָקוֹם שֶׁהֵם שְׂמֵחִים לִפְנֵי ה' - הָרֵי זֶה הַזְּמַן לְהַאֲזִין לִנְאוּמָיו שֶׁל הָרוֹעֶה הַגָּדוֹל, מֹשֶׁה בֶּן עַמְרָם. וְלָכֵן זֶה הַזְּמַן הַמַּתְאִים לְשִׂמְחַת תּוֹרָה שֶׁבִּכְתָב בִּכְלָל, שֶׁבְּסוּכּוֹת אָנוּ חוֹזְרִים לִהְיוֹת כְּמוֹ בְּנֵי יִשְׂרָאֵל דּוֹר מְקַבְּלֵי הַתּוֹרָה. שִׂמְחַת שְׁמִינִי עֲצֶרֶת הִיא שִׂמְחָה עַל הַשִּׂמְחָה, חָג עַל עֶצֶם זֶה שֶׁזָּכִינוּ עַכְשָׁו לַחְגֹּג שָׁבוּעַ יָמִים לִפְנֵי ה'. וַאֲנַחְנוּ מַמְשִׁיכִים לִקְרֹא בַּתּוֹרָה וְלִהְיוֹת אִתָּהּ כָּל הַשָּׁנָה, מַכְנִיסִים אוֹתָהּ אֵלֵינוּ הַבַּיְתָה.
סֵפֶר מִשְׁנֵה תוֹרָה, יֵשׁ בּוֹ בְּחִינַת "כָּל הַתּוֹרָה", כִּי הוּא הַשִּׂיא שֶׁל הֶסְבֵּר דֶּרֶךְ הַתּוֹרָה, מֹשֶׁה מְסַכֵּם לִבְנֵי יִשְׂרָאֵל אַחֲרֵי כָּל אַרְבָּעִים שְׁנוֹת לִמּוּד הַתּוֹרָה, מָה ה' אֱלֹהֶיךָ דּוֹרֵשׁ מֵעִמְּךָ, אַחֲרֵי שֶׁעָבְרוּ אֶת כָּל כּוּר הַבַּרְזֶל אַרְבָּעִים שָׁנָה, אֵלּוּ הַדְּבָרִים שֶׁהָיוּ צְרִיכִים לָקַחַת עִמָּם וּלְשַׁנֵּן בְּאָזְנֵיהֶם. וְדִבְרֵי מֹשֶׁה הָאַחֲרוֹנִים הֵם קְרִיאַת הַתּוֹרָה שֶׁאֲנַחְנוּ שׁוֹמְעִים שׁוּב וָשׁוּב בְּשִׂמְחַת תּוֹרָה.
לָמָּה צָרִיךְ לֶאֱכוֹל כָּשֵׁר?
בְּפָרָשַׁת הַשָּׁבוּעַ, פָּרָשַׁת שְׁמִינִי, מְפָרֶטֶת הַתּוֹרָה אֶת הָעוֹפוֹת הַטְּמֵאִים - הָאֲסוּרִים בַּאֲכִילָה. כֹּה נֶאֱמַר בְּפֶרֶק י"א:
"וְאֶת אֵלֶּה תְּשַׁקְּצוּ מִן הָעוֹף לֹא יֵאָכְלוּ שֶׁקֶץ הֵם: אֶת הַנֶּשֶׁר וְאֶת הַפֶּרֶס וְאֵת הָעָזְנִיָּה. וְאֶת הַדָּאָה וְאֶת הָאַיָּה לְמִינָהּ. אֵת כָּל עֹרֵב לְמִינוֹ. וְאֵת בַּת הַיַּעֲנָה וְאֶת הַתַּחְמָס וְאֶת הַשָּׁחַף; וְאֶת הַנֵּץ לְמִינֵהוּ. וְאֶת הַכּוֹס וְאֶת הַשָּׁלָךְ וְאֶת הַיַּנְשׁוּף. וְאֶת הַתִּנְשֶׁמֶת וְאֶת הַקָּאָת וְאֶת הָרָחָם. וְאֵת הַחֲסִידָה, הָאֲנָפָה לְמִינָהּ; וְאֶת הַדּוּכִיפַת, וְאֶת הָעֲטַלֵּף".
הַתּוֹרָה לֹא מְפָרֶטֶת אֶת סִימָנֵי הַטָּהֳרָה בָּעוֹפוֹת, אֶלָּא מְצַיֶּינֶת אֶת שְׁמוֹת הָעוֹפוֹת הַטְּמֵאִים. מִמֵּילָא, כָּל שְׁאָר הָעוֹפוֹת שֶׁאֵינָם מְפוֹרָטִים בְּפָסוּקֵי הַתּוֹרָה - הִינָם טְהוֹרִים וּמוּתָּרִים בַּאֲכִילָה. כָּךְ מוּבָא בְּסֵפֶר הַחִינּוּךְ (מִצְוָה קנז): "סִימָנֵי הָעוֹף לֹא נֶאֶמְרוּ, לְפִי שֶׁהַתּוֹרָה פֵּירְשָׁה לָנוּ בְּשֵׁם כָּל הַטְּמֵאִים שֶׁבָּעוֹלָם, וְכֵיוָן שֶׁכֵּן הוּא אֵין צוֹרֶךְ לִכְתוֹב לָנוּ סִימָנֵיהֶן, כִּי כָּל עוֹף הַנִּמְצָא בָּעוֹלָם חוּץ מֵהֶם הוּא כָּשֵׁר. וּמִכָּל מָקוֹם, מִפְּנֵי שֶׁכָּל אוֹתָן מִינִין שֶׁהוֹדִיעַתְנוּ הַתּוֹרָה אֵין כָּל אָדָם בָּקִי בָּהֶן, הוֹדִיעוּנוּ חֲכָמִים סִימָנִין בָּהֶן, כְּדֵי שֶׁיַּכִּיר אוֹתָן כָּל אָדָם וְיוּכַל לֶאֱכוֹל מִן הָעוֹפוֹת אַף עַל פִּי שֶׁאֵינוֹ מַכִּיר אֲפִילּוּ אֶחָד מִכָּל הַמִּינִין הַטְּמֵאִים הַמְּפֹרָשִׁים בַּכָּתוּב".
הָרַמְבַּ"ן מְבָאֵר כִּי כָּל הָעוֹפוֹת אוֹתָם אוֹסֶרֶת הַתּוֹרָה לֶאֱכוֹל הִינָם עוֹפוֹת טוֹרְפִים וְדוֹרְסִים. כֹּה דְּבָרָיו: "הַסִּימָן הַגָּדוֹל בָּעוֹפוֹת הִיא הַדְּרִיסָה, שֶׁכָּל עוֹף הַדּוֹרֵס לְעוֹלָם טָמֵא, כִּי הַתּוֹרָה הִרְחִיקַתְהוּ מִפְּנֵי שֶׁדָּמוֹ מְחוּמָּם לְאַכְזָרִיּוּתוֹ... וְנוֹתֵן אַכְזָרִיּוּת בַּלֵּב".
קוֹבֵעַ הָרַמְבַּ"ן: "וְהִנֵּה כָּל עוֹף שֶׁיִּדְרוֹס נֵדַע שֶׁהוּא מֵאֵלּוּ הַנִּזְכָּרִים, וְאִם בְּיָדוּעַ שֶׁאֵינוֹ דּוֹרֵס - וַדַּאי מוּתָּר". מַדּוּעַ אָסְרָה הַתּוֹרָה לֶאֱכוֹל עוֹפוֹת טוֹרְפִים? מְבָאֵר הָרַמְבַּ"ן: "וְהִנֵּה טַעַם הָאִיסּוּר בָּעוֹפוֹת, מִפְּנֵי אַכְזָרִיּוּת תּוֹלְדוֹתָם, וְהַבְּהֵמוֹת יִיתָּכֵן שֶׁיִּהְיוּ כֵּן, מִפְּנֵי שֶׁאֵין בְּבַעֲלֵי הַגֵּרָה וְהַפַּרְסָה הַשְּׁסוּעָה דּוֹרֵס, וְהַשְּׁאָר כּוּלָּם יִטְרוֹפוּ". רַבֵּנוּ בְּחַיֵּי מוֹסִיף עַל דִּבְרֵי הָרַמְבַּ"ן, כִּי כְּשֵׁם שֶׁבּוֹרֵא עוֹלָם אָסַר אֶת הַקְרָבַת הַקָּרְבָּן מֵעוֹפוֹת טְמֵאִים - כָּךְ אָסַר עַל עַמּוֹ, בָּנָיו אֲהוּבָיו, לְאוֹכְלָם.
כֹּה דְּבָרָיו: "וְהִנֵּה יֵשׁ לָתֵת טַעַם בְּאִיסּוּר הָעוֹפוֹת, כִּי לְפִי שֶׁהקב"ה בּוֹחֵר הַטּוֹב וְאוֹהֵב הַשִּׁפְלוּת וְכָל שְׁאָר הַמִּידּוֹת הַטּוֹבוֹת, וְשׂוֹנֵא הָאַכְזָרִיּוּת וְכָל שְׁאָר הַמִּידּוֹת הָרָעוֹת. לָכֵן בָּחַר בְּקָרְבָּנוֹ בַּבְּרִיּוֹת הַשְּׁפָלוֹת וְהַתְּמוֹת, וּמֵאוֹתָן שֶׁהוּא יִתְעַלֶּה בָּחַר בָּהֶן לְקָרְבָּן - צִיווָּה אֶת יִשְׂרָאֵל שֶׁיִּהְיֶה מַאֲכָלָם מֵהֶן, וּכְשֵׁם שֶׁהוּא יִתְבָּרַךְ מַרְחִיק הָעוֹפוֹת הַטְּמֵאִים לְצוֹרֶךְ קָרְבָּן, כֵּן מַרְחִיק אוֹתָם לְצוֹרֶךְ מַאֲכָלָם שֶׁל יִשְׂרָאֵל, וְכָל זֶה לְלַמְּדֵנוּ שֶׁהוּא יִתְעַלֶּה שׂוֹנֵא מִידַּת הָרַע וְהָאַכְזָרִיּוּת, וְאוֹהֵב מִידַּת הַטּוֹב וְהַשִּׁפְלוּת וְהַתְּמוֹת". כְּלוֹמַר, הָעוֹפוֹת הָאֲסוּרִים בַּאֲכִילָה הִינָם אַכְזָרִיִּים, וְכַאֲשֶׁר אוֹכְלִים מֵהֶם - נִכְנֶסֶת מִידַּת הָאַכְזָרִיּוּת בְּגוּפוֹ שֶׁל הָאוֹכֵל. מִשּׁוּם כָּךְ צִיווָּה עָלֵינוּ ה' יִתְבָּרַךְ לֹא לֶאֱכוֹל עוֹפוֹת טְמֵאִים, כְּדֵי שֶׁלֹּא תִּיכָּנֵס בְּדָמֵנוּ מִידַּת הָאַכְזָרִיּוּת. בְּאִיסּוּר זֶה מְלַמְּדֵנוּ בּוֹרֵא עוֹלָם כִּי "הוּא יִתְעַלֶּה שׂוֹנֵא מִידַּת הָרַע וְהָאַכְזָרִיּוּת, וְאוֹהֵב מִידַּת הַטּוֹב וְהַשִּׁפְלוּת וְהַתְּמוֹת".
שַׁבָּת שָׁלוֹם וּמְבוֹרָךְ, חַג שָׂמֵחַ, חוֹרֶף בָּרִיא וּגְאוּלָּה בְּרַחֲמִים אכי"ר.
לֶקֶט סִפּוּרִים...
פַּעַם אַחַת, בִּשְׁמִינִי-עֲצֶרֶת, נִכְנַס אִישׁ פָּשׁוּט לְרַבֵּנוּ הַזָּקֵן, רַבִּי שְׁנֵיאוֹר-זַלְמָן מִלַאדִי, וּבִיקְּשׁוֹ לְהַסְבִּיר לוֹ אֶת עִנְיַן שְׁמִינִי-עֲצֶרֶת בְּדֶרֶךְ שֶׁגַּם הוּא יוּכַל לְהָבִין. חִיכֵּךְ הָרַבִּי אֶת יָדוֹ עַל מִצְחוֹ וְאָמַר: "שְׁמִינִי-עֲצֶרֶת הוּא כְּמוֹ תַּמְצִית שֶׁמְּכִינִים בִּצְלוֹחִית קְטַנָּה, וְאַחַר-כָּךְ לוֹקְחִים מְעַט-מְעַט וּמְעָרְבְּבִים בְּמַיִם. כָּךְ שְׁמִינִי-עֲצֶרֶת הוּא כְּלָלוּת הַשִּׂמְחָה שֶׁל כָּל הַשָּׁנָה, וּמִמֶּנּוּ נִמְשֶׁכֶת הַשִּׂמְחָה בְּכָל יְמוֹת הַשָּׁנָה".
פַּעַם אַחַת, בִּשְׁמִינִי-עֲצֶרֶת, נִכְנַס אִישׁ פָּשׁוּט לְרַבֵּנוּ הַזָּקֵן, רַבִּי שְׁנֵיאוֹר-זַלְמָן מִלַאדִי, וּבִיקְּשׁוֹ לְהַסְבִּיר לוֹ אֶת עִנְיַן שְׁמִינִי-עֲצֶרֶת בְּדֶרֶךְ שֶׁגַּם הוּא יוּכַל לְהָבִין. חִיכֵּךְ הָרַבִּי אֶת יָדוֹ עַל מִצְחוֹ וְאָמַר: "שְׁמִינִי-עֲצֶרֶת הוּא כְּמוֹ תַּמְצִית שֶׁמְּכִינִים בִּצְלוֹחִית קְטַנָּה, וְאַחַר-כָּךְ לוֹקְחִים מְעַט-מְעַט וּמְעָרְבְּבִים בְּמַיִם. כָּךְ שְׁמִינִי-עֲצֶרֶת הוּא כְּלָלוּת הַשִּׂמְחָה שֶׁל כָּל הַשָּׁנָה, וּמִמֶּנּוּ נִמְשֶׁכֶת הַשִּׂמְחָה בְּכָל יְמוֹת הַשָּׁנָה".
"בַּיּוֹם הַשְּׁמִינִי עֲצֶרֶת תִּהְיֶה לָכֶם" (במדבר כט, לה). בְּרֹאשׁ-הַשָּׁנָה וּבְיוֹם-הַכִּיפּוּרִים הִתְפַּלַּלְנוּ בְּעַד הַנֶּפֶשׁ וְהִתְעַסַּקְנוּ בְּעִנְיְינֵי שָׁמַיִם בִּלְבַד. שְׁמִינִי-עֲצֶרֶת הוּא הַיּוֹם הָרִאשׁוֹן שֶׁבּוֹ מַתְחִילִים לְהִתְפַּלֵּל עַל הַגֶּשֶׁם, לִדְאוֹג לַצְּרָכִים הַגַּשְׁמִיִּים, לָעִנְיָינִים הַנּוֹגְעִים "לָכֶם" (רַבִּי משֶׁה-לֵייב מִסָּאסוֹב).
"בַּיּוֹם הַשְּׁמִינִי עֲצֶרֶת תִּהְיֶה לָכֶם" (במדבר כט, לה). בְּרֹאשׁ-הַשָּׁנָה וּבְיוֹם-הַכִּיפּוּרִים הִתְפַּלַּלְנוּ בְּעַד הַנֶּפֶשׁ וְהִתְעַסַּקְנוּ בְּעִנְיְינֵי שָׁמַיִם בִּלְבַד. שְׁמִינִי-עֲצֶרֶת הוּא הַיּוֹם הָרִאשׁוֹן שֶׁבּוֹ מַתְחִילִים לְהִתְפַּלֵּל עַל הַגֶּשֶׁם, לִדְאוֹג לַצְּרָכִים הַגַּשְׁמִיִּים, לָעִנְיָינִים הַנּוֹגְעִים "לָכֶם" (רַבִּי משֶׁה-לֵייב מִסָּאסוֹב).

שְׁמִינִי עֲצֶרֶת וְשִׂמְחַת תּוֹרָה תשל"ח, הָיוּ הֶחָגִים הַמְּתוּחִים בְּיוֹתֵר בְּקֶרֶב עֲדַת הַחֲסִידִים בְּבֵית חַיֵּינוּ, לְאַחַר שֶׁבַּעֲיצוּמָן שֶׁל הַהַקָּפוֹת בְּלֵיל שְׁמִינִי עֲצֶרֶת, עָבַר הָרַבִּי הֶתְקֵף לֵב קָשֶׁה. רוֹפְאִים הוּזְעֲקוּ לְחַדְרוֹ שֶׁל הָרַבִּי, אוּלָם הָרַבִּי סֵירֵב לִנְסוֹעַ לְבֵית הָרְפוּאָה, וְנִשְׁאַר בְּחַדְרוֹ. בֵּין לְבֵין, עוֹדֵד הָרַבִּי אֶת הַחֲסִידִים וְדָרַשׁ לְהוֹסִיף עוֹד וָעוֹד בַּשִּׂמְחָה. בְּמְלֹאת אַרְבָּעִים שָׁנָה לְאֵירוּעֵי הַיָּמִים, מֵבִיא 'בֵּית מָשִׁיחַ' שְׂבִיבִים מֵאוֹתָן 48 שָׁעוֹת שֶׁל מֶתַח וַחֲרָדָה.

בְּלֵיל שְׁמִינִי עֲצֶרֶת תשל"ח, הַכֹּל הִתְנַהֵל תְּחִילָּה כְּרָגִיל. תְּפִלַּת עַרְבִית הֵחֵלָּה בִּזְמַנָּהּ, וּבְשָׁעָה 9:00 נִכְנַס הָרַבִּי לְהַקָּפוֹת. הָיוּ הַרְבֵּה יוֹתֵר אוֹרְחִים מֵאֲשֶׁר בְּשָׁנִים קוֹדְמוֹת, וְהַקָּהָל נֶאֱמַד בְּכַמָּה אַלְפֵי אִישׁ. כַּאֲשֶׁר הָרַבִּי נִכְנַס רָקְדוּ וְשָׁרוּ. בְּכָל הַדֶּרֶךְ לַבִּימָה הוֹרָה בְּיָדוֹ לְהַגְבִּיר אֶת הַשִּׁירָה וּכְשֶׁעָלָה לַבִּימָה פָּנָה לְעֵבֶר הַקָּהָל הָרָחָב וּמָחָא כַּפַּיִם בְּחֹזֶק בְּמֶשֶׁךְ כַּמָּה דַּקּוֹת. אַחַר-כָּךְ הִתְחִילוּ בַּאֲמִירַת פְּסוּקֵי "אַתָּה הַרְאֵתָ" וְכַמָּה וְכַמָּה פְּעָמִים מָחָא כַּפַּיִם בְּחֹזֶק רַב לְעוֹדֵד אֶת הַשִּׁירָה.
הַקָּפָה רִאשׁוֹנָה עִם הרש"ג - עֲדַיִין כְּרָגִיל
אד"ש נִתְכַּבֵּד בְּסֵפֶר-תּוֹרָה הָרִאשׁוֹן (הַקָּטָן) בְּהַקָּפָה הָרִאשׁוֹנָה, וְיָצָא לְהַקָּפָה כְּשֶׁהוּא הוֹלֵךְ לְאַט וְאַחֲרָיו גִּיסוֹ הרש"ג וְזִקְנֵי אנ"ש. רַבִּים עָמְדוּ קָרוֹב לַשְּׁבִיל בּוֹ עָבַר אד"ש וְנִישְּׁקוּ אֶת הַסֵּפֶר-תּוֹרָה שֶׁבְּיָדוֹ כְּשֶׁהֵם מְאַחֲלִים "דֶּערְלֶעבְּט אִיבֶּער אַ יָאר", וְאד"ש עוֹנֶה: "גַּם אַתֶּם".
כְּשֶׁהִגִּיעַ הרש"ג, נֶעֱמַד אד"ש בָּאֶמְצַע וְהִמְתִּין עַד שֶׁיַּגִּיעַ סֵפֶר-תּוֹרָה שֶׁל מָשִׁיחַ. כְּשֶׁהִגִּיעַ וְנֶעֱמַד לְיַד הַשֻּׁלְחָן הִנִּיחַ אד"ש יָדוֹ הַיְמָנִית עַל כְּתֵפוֹ שֶׁל הרש"ג, שֶׁהִנִּיחַ אַף הוּא יָדוֹ עַל כֶּתֶף אד"ש, בְּיָד הַשְּׁנִיָּה הַסֵּפֶר-תּוֹרָה, וְהִתְחִילוּ לִרְקוֹד כְּשֶׁכָּל הַקָּהָל שָׁר אֶת נִיגּוּן הַהַקָּפוֹת. רָקְדוּ מֶשֶׁךְ זְמַן, וְאד"ש הָיָה בְּהִתְלַהֲבוּת יוֹתֵר מֵהָרָגִיל, עַד שֶׁהרש"ג נֶאֱלַץ לְהַפְסִיק וְאד"ש תָּמַךְ בּוֹ.
הַקָּפָה רְבִיעִית: הָרַבִּי מְבַקֵּשׁ אֶת הַכִּסֵּא
גַּם הַהַקָּפָה הַשְּׁנִיָּה וְהַשְּׁלִישִׁית הָיוּ עֲדַיִין כְּרָגִיל. רַק בְּהַקָּפָה הָרְבִיעִית, אד"ש הִסְתּוֹבֵב לְעֵבֶר הַקָּהָל וּמָחָא כַּפַּיִם לֹא בְּחֹזֶק - כַּפּוֹת הַיָּדַיִם בְּקוֹשִׁי נוֹגְעוֹת זוֹ בָּזוֹ - כְּשֶׁיָּדוֹ הַשְּׂמָאלִית נִשְׁעֶנֶת עַל הַסְּטֶנְדֶּר, לְמֶשֶׁךְ זְמַן קָצָר לְפִי עֵרֶךְ. פָּנָיו שֶׁל אד"ש נִשְׁתַּנּוּ לְחִיוְורִים כְּסִיד וּפָנָה אֶל הַמַּזְכִּיר הרב גרונר וְאָמַר: "גִיב מִיר מַיין בֵּיינְקְל" [הָבֵא לִי אֶת הַכִּסֵּא]. אד"ש הִתְיַישֵּׁב וְקֵירֵב אֵלָיו אֶת הַסִּידּוּר. כְּשֶׁיָּשַׁב נִשְׁעַן עַל הַמִּשְׁעֶנֶת וְעָצַם עֵינָיו וְנִרְאָה הָיָה שֶׁסּוֹבֵל כְּאֵבִים.
מַיִם, מַיִם!
הרב גרונר הִזְעִיק אֶת אֶחָד הָרוֹפְאִים שֶׁעָמַד לְיַד הַבִּימָה וְהַד"ר אָמַר שֶׁיָּבִיאוּ מַיִם. צָעֲקוּ שֶׁיָּבִיאוּ מַיִם. הַמַּזְכִּיר הרב בנימין קליין הוֹשִׁיט לְאד"ש סֵפֶל מַיִם אַךְ אד"ש לֹא רָצָה לִשְׁתּוֹת, וּבִכְלָל לֹא רָצָה שֶׁהָרוֹפֵא יִבְדּוֹק אֶת הַדּוֹפֶק. הַקָּהָל הָיָה בְּמֶתַח הִיסְטֵרִי. צָעֲקוּ לְהִתְרַחֵק מֵהַבִּימָה וְלָצֵאת מִבֵּיהַכְנֶ"ס כְּדֵי שֶׁיִּהְיֶה אֲוִויר יוֹתֵר מְרֻוָּוח. תּוֹךְ דַּקּוֹת אֲחָדוֹת הִתְרוֹקֵן כִּמְעַט כָּל בֵּיהַכְנֶ"ס. נִשְׁאֲרוּ פָּחוֹת מִמֵּאָה אִישׁ. גַּם עֶזְרַת נָשִׁים הִתְרוֹקְנָה וְנִפְרְצוּ הַחַלּוֹנוֹת שָׁם בִּכְדֵי שֶׁיִּכָּנֵס דַּרְכָּם אֲוִויר צַח וְקָרִיר.
הַקָּפָה חֲמִישִׁית וְשִׁישִּׁית - מְהִירוּת
לְהַקָּפָה הַחֲמִישִׁית קָם אד"ש מִכִּסְאוֹ וּמָחָא כַּפַּיִם מֶשֶׁךְ זְמַן קָצָר (לְפִי נוּסְחָא אַחַת, נֶעֶרְכָה גַּם הַהַקָּפָה הַזּוֹ - כְּמוֹ הַשִּׁישִּׁית - בִּמְהִירוּת, לְלֹא שִׁירָה וְרִיקּוּד) וְאַחֲרֶיהָ הִתְיַישֵּׁב. אֶת הַהַקָּפָה הַשִּׁישִּׁית עָרְכוּ בְּשֶׁקֶט, בְּלִי שִׁירָה וְרִיקּוּד, כְּדֵי שֶׁאד"ש לֹא יִצְטָרֵךְ לָקוּם.
הַקָּפָה הַשְּׁבִיעִית - עִם הרש"ג
לְהַקָּפָה הַשְּׁבִיעִית קָם אד"ש מִכִּסְאוֹ. הִצִּיעוּ שֶׁיַּעַרְכוּ אֶת הַהַקָּפָה לְיַד הַבִּימָה שֶׁלּוֹ אַךְ אד"ש מֵיאֵן, קִיבֵּל אֶת הַסֵּפֶר-תּוֹרָה הַקָּטָן מִיָּדֵי הרב בנימין אלטהויז וְהָלַךְ עִמּוֹ, בְּאִיטִּיּוּת וּבְפָנִים חִיוְורוֹת, לִמְקוֹם הַהַקָּפוֹת בְּמֶרְכַּז בֵּיהַכְנֶ"ס. הָיוּ כָּאֵלּוּ שֶׁרָצוּ לְנַשֵּׁק אֶת הַסֵּפֶר-תּוֹרָה אַךְ הַמַּזְכִּיר הרב גרונר, שֶׁצָּעַד לִפְנֵי אד"ש, מָנַע זֹאת מֵהֶם כְּדֵי שֶׁיִּסְתַּיֵּים מַהֵר. אַחֲרֵי אד"ש הָלַךְ הרש"ג לְבִימַת הַהַקָּפוֹת. אד"ש הֵחֵל לִרְקוֹד עִמּוֹ מְעַט אַךְ הרש"ג הִפְסִיק מִיָּד. כְּשֶׁהִפְסִיק חִיֵּיךְ אד"ש וְעָשָׂה תְּנוּעָה שֶׁל תְּמִיהָה. הַהַקָּפָה הַזּוֹ נִמְשְׁכָה בְּעֵרֶךְ דַּקָּה. הָיָה נִרְאֶה כְּאִילּוּ אד"ש צָרִיךְ שֶׁיֹּאחֲזוּ בּוֹ וְהרב גרונר נִיסָּה לַעֲשׂוֹת זֹאת, אֲבָל אד"ש לֹא הִסְכִּים.
הָרַבִּי עוֹלֶה לְחַדְרוֹ
אַחֲרֵי הַתְּפִילָּה אָמַר ג' פְּעָמִים "גוּט יוֹם טוֹב" (בְּקוֹל דַּי חַלָּשׁ) כְּשֶׁהוּא מֵנִיף שְׁתֵּי יָדָיו וּמֵנִיעַ כָּל גּוּפוֹ. אַחַר-כָּךְ הִתְחִיל לָשִׁיר "וְשָׂמַחְתָּ" וּבְדַרְכּוֹ לָצֵאת מִבֵּיהַכְנֶ"ס רָמַז בְּיָדוֹ לְהַגְבִּיר אֶת הַשִּׁירָה. כְּשֶׁהִגִּיעַ לַמַּדְרֵגוֹת רָצוּ לַעֲזוֹר לוֹ אַךְ דָּחָה אֶת הָעֶזְרָה. כְּשֶׁנִּכְנַס אד"ש לְחַדְרוֹ נִשְׁעַן פִּתְאוֹם עַל הַקִּיר. הרב חודקוב שֶׁהָיָה עִמּוֹ פָּחַד מִזֶּה וְאָמַר לוֹ אד"ש שֶׁיֵּלֵךְ לְ"מֶרְכָּז" (חֶדֶר הַמַּזְכִּירוּת) וְלֹא יִשָּׁאֵר בְּחַדְרוֹ.
הָרַבִּי פּוֹתֵחַ וּמַרְגִּיעַ
אַחֲרֵי זְמַן יָרְדָה מִקּוֹמָה שְׁלִישִׁית גִּיסָתוֹ, אֵשֶׁת הרש"ג, וְדָפְקָה עַל הַדֶּלֶת, וּכְשֶׁלֹּא נִפְתְּחָה מִיָּד דָּפְקָה חָזָק וְצָעֲקָה אֶת שֵׁם אד"ש (הַשֵּׁנִי) פַּעֲמַיִים "... עֶפֶען אוֹיף" [תִּפְתַּח]. אד"ש פָּתַח וְהִיא שָׁאֲלָה "מָה נִשְׁמָע?". אד"ש אָמַר לָהּ שֶׁהַשָּׁבוּעַ לֹא יָשַׁן וְכָל הַיּוֹם עָמַד וְחִילֵּק לֶעקָאח, "אִיךְ הָאבּ נִיט געגעסען אַ גַאנְצֶן טָאג און נִיט געשלָאפֶן" [לֹא אָכַלְתִּי כָּל הַיּוֹם, וְלֹא יָשַׁנְתִּי], אֲבָל הַלַּיְלָה יָנוּחַ וּבְעֶזְרַת הַשֵּׁם מָחָר יִהְיֶה סֵדֶר רָגִיל.
קִידּוּשׁ בַּסוּכָּה
הָרוֹפְאִים בִּיקְּשׁוּ שֶׁיֹּאכַל וְיִשְׁתֶּה בְּחַדְרוֹ אַךְ אד"ש שָׁלַל זֹאת בְּאוֹמְרוֹ שֶׁלֹּא יֹאכַל לִפְנֵי קִידּוּשׁ וּמִחוּץ לַסוּכָּה. כְּשֶׁאֶחָד הָרוֹפְאִים אָמַר שֶׁהַהֲלִיכָה לַסוּכָּה תַּקְשֶׁה עָלָיו, אָמַר אד"ש: אֵיךְ יָכוֹל אֲנִי לְהַשְׁאִיר אֶת הָרַבָּנִית לְבַדָּהּ כָּל הַלַּיְלָה, אִם לֹא אֶעֱשֶׂה קִידּוּשׁ עֲבוּרָהּ הִיא לֹא תֹּאכַל. בְּחַדְרוֹ שָׁהָה אד"ש כְּעֶשֶׂר דַּקּוֹת וְאַחַר-כָּךְ יָצָא מִשָּׁם, לָבוּשׁ מְעִיל, בִּכְדֵי לְהִכָּנֵס לַסוּכָּה. הַקָּהָל עָמַד מֵרָחוֹק, מֵעֵבֶר לְ"סֶרְוִויס-לַיין", וּכְשֶׁנִּרְאָה אד"ש יוֹצֵא הִתְחִילוּ לָשִׁיר. אַחַר-כָּךְ נִכְנְסָה לַסוּכָּה הָרַבָּנִית (הִיא הָיְתָה בַּבַּיִת שֶׁבִּרְחוֹב פְּרֶזִידֶנְט וּבְמַהֲלַךְ הַהַקָּפוֹת רָצוּ לְהוֹדִיעַ לָהּ. לְ-770 הִגִּיעָה דֶּרֶךְ רְחוֹב יוּנְיוֹן).
קִידּוּשׁ עַל יַיִן דַּווְקָא
לַסוּכָּה נִכְנְסוּ גַּם הָרוֹפְאִים וְאֶחָד הַמַּזְכִּירִים. כְּשֶׁנִּכְנַס אד"ש לַסוּכָּה יָשַׁב וְהִשְׁעִין רֹאשׁוֹ עַל יָדוֹ. הִכְנִיסוּ לְשָׁם אֹכֶל וּמִיץ-עֲנָבִים אַךְ אד"ש אָמַר: "קִידּוּשׁ מַאכְט מֶען אוֹיף וַויין, נִיט אוֹיף גְרֵייפּ-דְשׁוּס!". אד"ש קִידֵּשׁ עַל יַיִן וְאַחֲרֵי זֶה אָכַל לֶעקָאח. אַחַר-כָּךְ נָטַל יָדָיו וְאָכַל חַלָּה אוֹתָהּ טָבַל בְּמֶלַח שֶׁהָיָה נָתוּן בְּכַף עַל הַשֻּׁלְחָן. אָכַל סְעוּדָּה קְטַנָּה וּבֵירַךְ בִּרְכַּת הַמָּזוֹן. בִּיצִיאָתוֹ, עוֹדֵד בְּיָדוֹ אֶת שִׁירַת הַקָּהָל.
הָרַבִּי מְבַקֵּשׁ: לִהְיוֹת רְגוּעִים, וּבְשִׂמְחָה
כַּאֲשֶׁר יָשַׁב אד"ש בַּסוּכָּה הִכְנִיסוּ לְחַדְרוֹ מִיטָּה מִדִּירַת אַדְמוּ"ר (מהוריי"צ) נ"ע שֶׁבְּקוֹמָה הַשְּׁנִיָּיה. אד"ש פָּתַח אֶת דֶּלֶת חַדְרוֹ וּבִיקֵּשׁ לְהוֹדִיעַ לַקָּהָל שֶׁיִּהְיוּ רְגוּעִים, וְיֵלְכוּ לַחְגֹּג שִׂמְחַת יוֹם טוֹב, וְשֶׁאֵלּוּ שֶׁעֲדַיִין לֹא עָשׂוּ הַקָּפוֹת יֵלְכוּ לַעֲרֹךְ הַקָּפוֹת בְּשִׂמְחָה. הַמַּזְכִּיר הרב גרונר יָצָא וְהוֹדִיעַ זֹאת לַקָּהָל. כֵּן הוֹדִיעַ אד"ש שֶׁהַלַּיְלָה לֹא יֵלֵךְ הַבַּיְתָה.
הָרַבִּי לֹא מַסְכִּים לִנְסוֹעַ לְבֵית רְפוּאָה
לְעֵרֶךְ בְּשָׁעָה אַחַת אַחַר חֲצוֹת אָמַר אד"ש שֶׁיֵּשׁ לוֹ כְּאֵבִים בַּלֵּב (עַד אָז לֹא אָמַר זֹאת). הִגִּיעוּ רוֹפְאִים מוּמְחִים בְּתְחוּם זֶה. הָיוּ שֶׁהִצִּיעוּ שֶׁיִּיסַּע לְבֵית-רְפוּאָה אַךְ אד"ש לֹא הִסְכִּים בְּשׁוּם אוֹפֶן וְשָׁאַל מָה עוֹשֶׂה אָדָם הַנִּמְצָא בְּמָקוֹם שֶׁאֵין בִּסְבִיבָתוֹ בֵּית-רְפוּאָה? אד"ש הוֹדִיעַ לָרוֹפְאִים שֶׁהוּא אֵינוֹ נוֹסֵעַ אֶלָּא נִשְׁאָר לָנוּחַ בְּחַדְרוֹ.
הָרוֹפְאִים מִתְפַּלְּאִים
בְּדִיקוֹת הָרוֹפְאִים הֶעֱלוּ שֶׁאד"ש עָבַר הֶתְקֵף לֵב קָשֶׁה בְּיוֹתֵר. מְסַפְּרִים שֶׁאֶחָד הָרוֹפְאִים שָׁאַל אֶת הַמַּזְכִּירִים הַאִם בְּעֵת הַהֶתְקֵף - בִּשְׁעַת הַהַקָּפוֹת - נֶאֱנַק אד"ש מִכְּאֵבִים, וּכְשֶׁנַּעֲנָה בַּשְּׁלִילָה וְשָׁמַע שֶׁרַק יָשַׁב עַל כִּסְאוֹ, אָמַר הָרוֹפֵא שֶׁזֶּה לוֹ כְּבָר עֲשָׂרוֹת שָׁנִים בְּתַפְקִידוֹ וּמֵעוֹלָם לֹא שָׁמַע עַל תּוֹפָעָה כָּזוֹ, שֶׁכֵּן מְדֻבָּר בְּכְאֵב שֶׁהוּא לְמַעְלָה מִכּוֹחַ הַסֶּבֶל הָאֱנוֹשִׁי.
הָרַבִּי: הָרְפוּאָה תָּבוֹא מִכַּאן!
הַדּוֹקְטוֹר מָרְדֳּכַי מְנַחֵם מֶענְדֶל גְלֶזְמָן סִיפֵּר עַל דּוּ-שִׂיחַ שֶׁהָיָה לוֹ עִם הָרַבִּי בַּנּוֹשֵׂא: "אָמַרְתִּי לָרַבִּי שֶׁבְּעִנְיְינֵי תּוֹרָה וְיַהֲדוּת אֲנִי מְצַיֵּית לָרַבִּי, אֲבָל בְּעִנְיְינֵי רְפוּאָה אֲנִי מְבַקֵּשׁ שֶׁהָרַבִּי יִשְׁמַע לְדַעְתִּי וְיֵלֵךְ לְבֵית הָרְפוּאָה. הָרַבִּי הֵשִׁיב לִי: 'בְּאוֹפֶן עִקְרוֹנִי אַתָּה צוֹדֵק, אֲבָל בַּמִּקְרֶה הַסְּפֶּצִיפִי שֶׁלִּי, אַתָּה לֹא יוֹדֵעַ אֶת כָּל הַנְּתוּנִים וּבְמֵילָא אֵינְךָ יָכוֹל לְהַחְלִיט'.
"שָׁאַלְתִּי אֶת הָרַבִּי: 'מָה זֹאת אוֹמֶרֶת "לֹא יוֹדֵעַ אֶת כָּל הַנְּתוּנִים"? מָה אֲנִי צָרִיךְ לָדַעַת? הֲרֵי הָרַבִּי לֹא לָקַח עַד הַיּוֹם תְּרוּפוֹת. זֶה דָּבָר שֶׁפִּתְאוֹם קָרָה'". הָרַבִּי אָמַר 'לֹא!' בְּהַחְלָטִיּוּת, וְהוֹסִיף: 'אֲנִי מַעֲדִיף לְהִישָּׁאֵר כָּאן. יֵשׁ לְךָ מוּשָּׂג מָה קָרָה כָּאן בַּחֶדֶר הַזֶּה, וְעַל הַשֻּׁלְחָן שֶׁלִּי?' הוֹסִיף וְשָׁאַל הָרַבִּי לְפֶתַע. "אֲנִי לֹא יוֹדֵעַ, אֲבָל אֲנִי מְשַׁעֵר", עָנִיתִי". "אֲנִי לֹא יָכוֹל אֲפִילּוּ לְהַתְחִיל לְסַפֵּר לְךָ מָה עָבַר בַּשֻּׁלְחָן הַזֶּה", אָמַר הָרַבִּי, "אֲבָל בִּשְׁבִילִי הַחֶדֶר הַזֶּה הוּא קוֹדֶשׁ קוֹדָשִׁים, וְאַתָּה אוֹמֵר לִי עַכְשָׁו לָצֵאת מִקֹּדֶשׁ הַקּוֹדָשִׁים לְבֵית הָרְפוּאָה?... 'הָרְפוּאָה תָּבוֹא מִכָּאן', הוֹסִיף הָרַבִּי וְאָמַר לְאַחַר שְׁהוּת קַלָּה..."
הָרוֹפְאִים מְבַקְּשִׁים מֵהָרַבָּנִים לִפְסוֹק
הָיוּ רוֹפְאִים שֶׁאָמְרוּ שֶׁיֵּשׁ לָקַחַת אֶת אד"ש לְבֵית-הָרְפוּאָה אַף שֶׁלֹּא בְּהַסְכָּמָתוֹ, בְּעֵת הַשֵּׁינָה. קָרְאוּ לָרַבָּנִים לָדוּן בַּדָּבָר אַךְ קָשְׁתָה עֲלֵיהֶם הַהַכְרָעָה. קוֹדֶם לָכֵן הֵכִינוּ עַצְמָם הָרַבָּנִים שֶׁאִם יִשְׁאֲלוּ אֶצְלָם יֹאמְרוּ שֶׁצָּרִיךְ לִשְׁמוֹעַ לָרוֹפְאִים.
הָרַבִּי מְבַקֵּשׁ שֶׁהָרַבָּנִים לֹא יִפְסְקוּ
כְּשֶׁאד"ש הִתְעוֹרֵר אָמַר שֶׁבְּשׁוּם אוֹפֶן לֹא יֵלֵךְ לְבֵית-הָרְפוּאָה וְשֶׁעַל הָרוֹפְאִים יֵשׁ לִשְׁקוֹל מוּל הַמַּעֲלוֹת שֶׁל הַשְּׁהִיָּיה בְּבֵית-רְפוּאָה גַּם אֶת הַחֶסְרוֹן שֶׁבְּאִי-הַנְּעִימוּת שֶׁתִּהְיֶה לוֹ אִם יֵלֵךְ לְשָׁם נֶגֶד רְצוֹנוֹ, וּלְהַכְרִיעַ שֶׁטּוֹב יוֹתֵר שֶׁיִּישָּׁאֵר בְּחַדְרוֹ. לְפִי נוּסְחָא אַחַת גַּם אָמַר שֶׁאִם יִהְיֶה בְּבֵית-רְפוּאָה הַדָּבָר יְעוֹרֵר בֶּהָלָה אֵצֶל הַחֲסִידִים, וְכֵן אֲשֶׁר כָּאן בְּחַדְרוֹ הוּא בַּעַל-הַבַּיִת, וְאִילוּ בְּבֵית הָרְפוּאָה הוּא אֶחָד בֵּין הַרְבֵּה חוֹלִים. כְּשֶׁרָצוּ הָרַבָּנִים לְהִכָּנֵס לְחַדְרוֹ - הוֹרָה הָרַבִּי לִמְסוֹר לָהֶם: הֱיוֹת וַאֲנִי "אַ פְרוּמֶער אִיד" וְצָרִיךְ לִשְׁמוֹעַ בְּקוֹל הָרַבָּנִים, מִמֵּילָא אֲנִי מְבַקֵּשׁ שֶׁלֹּא יֹאמְרוּ לִי כָּל פְּסָק כְּדֵי שֶׁלֹּא אֵאָלֵץ לָלֶכֶת, שֶׁכֵּן אֵין צוֹרֶךְ שֶׁאֵלֵךְ".
הָרַבָּנִית מַכְרִיעָה
נִיסּוּ לְבַקֵּשׁ מֵהָרַבָּנִית שֶׁתְּבַקֵּשׁ מֵאד"ש שֶׁיַּסְכִּים לִנְסוֹעַ, אַךְ הִיא אָמְרָה: "הוּא יוֹדֵעַ בְּעַצְמוֹ מָה לַעֲשׂוֹת", וְהוֹסִיפָה כִּי מֵעוֹלָם לֹא אָמְרָה מַשֶּׁהוּ בְּנִיגּוּד לִרְצוֹנוֹ שֶׁל הָרַבִּי.
הרש"ג: לְצַיֵּית לְדִבְרֵי הָרַבִּי
בְּשָׁלָב מְסוּיָּם בִּיקְּשָׁה הָרַבָּנִית מֵאֶחָד הַמשב"קים לַעֲלוֹת לִשְׁאוֹל לְדַעְתּוֹ שֶׁל הרש"ג. בְּעוֹדוֹ עוֹלֶה לַקּוֹמָה הַשְּׁלִישִׁית, עָלְתָה הָרַבָּנִית אַחֲרָיו, וְכַאֲשֶׁר הרש"ג שָׁמַע אֶת הַשְּׁאֵלָה, בִּיקֵּשׁ לָדַעַת מָה הָרַבִּי עַצְמוֹ אוֹמֵר. לְאַחַר שֶׁשָּׁמַע כִּי הָרַבִּי מְסָרֵב לַעֲזֹב לְבֵית הָרְפוּאָה, אָמַר: אִם הָרַבִּי לֹא רוֹצֶה, כָּךְ צָרִיךְ לַעֲשׂוֹת!
תְּפִילָּה בְּגַן עֵדֶן הַתַּחְתּוֹן
בִּשְׁמִינִי עֲצֶרֶת בַּבּוֹקֶר, הִתְפַּלְּלוּ תְּפִילַּת שַׁחֲרִית בְּמִנְיָין מְצוּמְצָם בְּגַן עֵדֶן הַתַּחְתּוֹן וּקְרִיאַת-הַתּוֹרָה הִתְקַיְּימָה בְּחַדְרוֹ שֶׁל אד"ש עַל הַסְּטֶנְדֶּר שֶׁל אד"ש. אד"ש עָלָה לַתּוֹרָה וְאָמַר אֶת הַהַפְטָרָה בִּישִׁיבָה. אֶת הַהַפְטָרָה יָכְלוּ לִשְׁמוֹעַ גַּם בְּגַן-עֵדֶן-הַתַּחְתּוֹן, שָׁם הִתְפַּלְּלוּ תְּפִילַּת מוּסָף. דֶּלֶת הַחֶדֶר נִשְׁאֲרָה פְּתוּחָה. אַחֲרֵי הַהַפְטָרָה שָׁכַב שׁוּב בַּמִּיטָּה. בְּ"הוּא אֱלוֹקֵינוּ" וּבְ"אַתָּה בְּחַרְתָּנוּ" רָמַז בְּיָדוֹ שֶׁיֵּשׁ לָשִׁיר עַל יְדֵי שֶׁהִכָּה קַלּוֹת עַל הַמִּטָּה. אַחֲרֵי הַתְּפִילָּה הִתְחִיל לָשִׁיר "וְשָׂמַחְתָּ בְּחַגֶּךָ" וְאַחַר-כָּךְ בִּיקֵּשׁ שֶׁכָּל הַמִּתְפַּלְלִים יַעַבְרוּ לְפָנָיו וּבֵירֵךְ אֶת כּוּלָּם בְּ"גוּט יוֹם טוֹב". לַכֹּהֲנִים אָמַר "יִישַׁר כּוֹחַ". לָרַב שְׁמַערְל גוּרָארי' (מִתֵּל-אָבִיב) הוֹרָה לְדַבֵּר עִם הַגַּבָּאִים אוֹדוֹת מְכִירַת הַמִּצְווֹת בְּשַׁבָּת בְּרֵאשִׁית.
הַדּוֹקְטוֹר וַוייס מַגִּיעַ
בְּמַהֲלַךְ הַתְּפִילָּה הִגִּיעַ הַד"ר וַוייס מִשִּׁיקָגוֹ, שֶׁאַף הֵבִיא עִמּוֹ כַּמָּה מַכְשִׁירִים מִבֵּית-הָרְפוּאָה שֶׁלּוֹ, שֶׁבְּאֶמְצָעוּתָם יוּכְלוּ שָׁם לִבְדֹּק אֶת תּוֹצְאוֹת הַבְּדִיקוֹת שֶׁיַּעֲשֶׂה כָּאן. מְסַפְּרִים שֶׁכַּאֲשֶׁר הִגִּיעַ לְ-770 פָּרַץ בִּבְכִי וְאָמַר: תָּמִיד יָדַעְנוּ שֶׁאד"ש מְחוֹלֵל נִסִּים וְנִפְלָאוֹת לַחוֹלִים וְכוּ', וַאֲנִי בְּעַצְמִי נוֹכַחְתִּי בְּכָךְ כְּשֶׁטִּיפַּלְתִּי בְּכַמָּה רוֹפְאִים שֶׁאד"ש שָׁלַח אֵלַי לְהִתְרַפְּאוּת, אַךְ כָּעֵת עָלַי לָבוֹא וּלְרַפְּאוֹת אֶת אד"ש. כְּשֶׁנִּכְנַס לַחֶדֶר אָמַר לוֹ אד"ש שֶׁרוֹצֶה שֶׁיְּטַפֵּל בּוֹ כְּמוֹ בְּכָל פַּצְיֶנְט רָגִיל.
מְסַפְּרִים שֶׁעֶצֶם הַגָּעָתוֹ שֶׁל הַד"ר וַוייס גָּרְמָה לְאד"ש לִהְיוֹת בְּמַצַּב-רוּחַ מְרוֹמָם. הַד"ר וַוייס אָמַר שֶׁאָמְנָם קָשֶׁה לְטַפֵּל בְּחוֹלֶה כְּשֶׁאֵינוֹ בְּבֵית-הָרְפוּאָה, אֲבָל זֶה אֶפְשָׁרִי. עַל עַצְמוֹ אָמַר שֶׁאֵינוֹ רוֹפֵא גָּדוֹל, אֲבָל יוֹדֵעַ מַהוּ "רַבִּי" וְלָכֵן מְקַוֶּוה שֶׁיּוּכַל לָתֵת אֶת הַטִּפּוּל הַטּוֹב בְּיוֹתֵר. הוּא אָמַר שֶׁיִּישָּׁאֵר לְיַד אד"ש עַד אֲשֶׁר יַבְרִיא. הוּא אַף אָמַר שֶׁיֵּשׁ לִשְׁמוֹעַ לְדַעַת אד"ש וּלְאַפְשֵׁר לוֹ לְהִשָּׁאֵר בְּחַדְרוֹ, וּמָנָה כַּמָּה מַעֲלוֹת בְּכָךְ: כָּאן יִהְיֶה הַטִּפּוּל עִם רוֹפֵא צָמוּד, אד"ש יוּכַל לְהַבִּיעַ דַּעְתּוֹ עַל הַטִּפּוּל וְגַם לֹא יִצְטַעֵר מֵהָרִיחוּק מֵהַחֲסִידִים.
נְסִיעָה לְקְראוּן הַייטְס, עִם סְגַן הַמְּפַכ"ל
לְיָמִים סִיפֵּר הַד"ר וַוייס, שֶׁכְּשֶׁנָּחַת בִּשְׂדֵה הַתְּעוּפָה בְּנְיוּ יוֹרְק, לֹא הָיָה לוֹ מוּשָּׂג הֵיכָן שְׁכוּנַת קְראוּן הַייטְס. הוּא שׁוֹטֵט בִּנְמַל הַתְּעוּפָה, וּלְפֶתַע נִיגַּשׁ אֵלָיו שׁוֹטֵר מִמִּשְׁטְרַת נְיוּ יוֹרְק וְשׁוֹאֵל הַאִם הוּא ד"ר אַיִירָה וַוייס מִשִּׁיקָאגוֹ? לְאַחַר שֶׁהֵשִׁיב בְּחִיּוּב, וְאָמַר שֶׁהוּא זָקוּק לְעֶזְרָה, הֵגִיב הַשּׁוֹטֵר וְאָמַר שֶׁהַשּׁוֹטְרִים הִתְבַּקְּשׁוּ לִמְצֹא אוֹתוֹ, וְהֵם מְקַבְּלִים אֶת פְּנֵי הַנּוֹסְעִים בְּכָל מָטוֹס הַמַּגִּיעַ מִשִּׁיקָאגוֹ, כְּשֶׁהֵם מְחַפְּשִׂים אָדָם שֶׁיָּכוֹל לִהְיוֹת ד"ר וַוייס. סְגַן מְפַכ"ל הַמִּשְׁטָרָה הִסִּיעַ אוֹתוֹ בְּשַׁיָּירָה לְקְראוּן הַייטְס, וְכָךְ הִגִּיעַ לְ-770 בִּמְהִירוּת שִׂיא.
הָרַבָּנִית מַרְגִּיעָה: "אַל דְּאָגָה, ד"ר וַוייס"
עַל פְּגִישָׁתוֹ עִם הָרַבָּנִית, סִיפֵּר הַד"ר וַוייס: חָצִיתִי אֶת יָם הָאֲנָשִׁים, וְנִכְנַסְתִּי פְּנִימָה. אֲבָל אָז, נֶעֱצַרְתִּי עַל יְדֵי הָרַבָּנִית שְׁנֵיאוּרְסֹאהְן, אִשָּׁה עֲדִינָה וְאָדִיבָה בְּיוֹתֵר. אִשָּׁה פָּשׁוּט מַדְהִימָה. לֹא רָצִיתִי לִהְיוֹת בִּלְתִּי מְנוּמָּס, אֲבָל אָמַרְתִּי לָהּ: רֶעבֶּצִין, הָיִיתִי שָׂמֵחַ לְהַכִּירֵךְ, אוּלָם אֲנִי הָרֶגַע הִגַּעְתִּי מִשִּׁיקָגוֹ וּבִרְצוֹנִי לָרוּץ לְהִכָּנֵס )לָרַבִּי( מִיָּד. "אַל דְּאָגָה, ד"ר וַוייס", אָמְרָה הָרַבָּנִית חַיָּה מוּשְׁקָא, "יְדִידְךָ, ד"ר טִישְׁהוֹלְץ הִגִּיעַ לְכָאן מִיָּד, וְהוּא סִידֵּר אֶת הָעִנְיָינִים". "בַּעֲלִי )'מַיין מַאן'( מְדַבֵּר, לַחַץ דָּמוֹ הִתְיַצֵּב, וְנִרְאֶה שֶׁהוּא נִמְצָא בְּמַצָּב חָדָשׁ".
סוֹכְנוּת הַחֶלֶל מִתְגַּיֶּיסֶת לְמַעַן הָרַבִּי
ד"ר טִישְׁהוֹלְץ, שֶׁהָיוּ לוֹ קְשָׁרִים רַבִּים בְּנְיוּ יוֹרְק, בֵּינֵיהֶם עִם נאס"א )סוֹכְנוּת הַחֶלֶל הָאֲמֵרִיקָאִי(, הִצְלִיחַ לְהָבִיא לְ-770 עֲבוּר הָרַבִּי מַכְשִׁיר נִיטוּר לֵב אַלְחוּטִי. בַּיָּמִים הָהֵם, מִי שֶׁנִּזְקַק לְנִיטוּר לֵב הָיָה מְרוּתָק לְמִיטָּה וּמְחֻבָּר לְמַכְשִׁיר מְסֻרְבָּל עִם חוּטֵי חַשְׁמַל רַבִּים, אַךְ הוּא הִצְלִיחַ לְהַשִּׂיג אֶת הַמַּכְשִׁיר הָאַלְחוּטִי מִבָּסִיס נאס"א.
מַחְלֶקֶת הַמְּדִינָה בְּאַרה"ב מְסַיַּיעַת
אֶחָד הָרוֹפְאִים הָיָה ד"ר בֶּרְנַרְד לֵאוֹן. לְיָמִים סִיפֵּר הַד"ר וַוייס, כִּי לֵאוֹן עָסַק בְּאוֹתָהּ עֵת בְּפְרוֹיֶקְט מֶמְשַׁלְתִּי עִם מְדִינַת סִין, וּמַחְלֶקֶת הַמְּדִינָה שֶׁל אַרה"ב הֵבִיאָה אוֹתוֹ בְּמְיוּחָד כְּדֵי לְטַפֵּל בָּרַבִּי. פְּקִידִים בְּמַחְלֶקֶת הַמְּדִינָה הִכִּירוּ בָּרַבִּי כְּמֻמְחֶה בְּבִיטָּחוֹן מְדִינִי, וְנָהֲגוּ לְהִתְיָיעֵץ אִתּוֹ. הָיָה חָשׁוּב לָהֶם שֶׁהָרַבִּי יִהְיֶה בָּרִיא.
הָרַבִּי: "לָלֶכֶת לְשַׂמֵּחַ יְהוּדִים!"
אַחַר הַצָּהֳרַיִם הִתְקַבְּלָה הוֹרָאָה מֵאד"ש שֶׁכּוּלָּם יֵלְכוּ לְבָתֵּי-כְּנֵסִיּוֹת וִישַׂמְּחוּ יְהוּדִים "מִיט אַ שְׁטוֹרֶעם". אד"ש אָמַר שֶׁאֵלּוּ שֶׁרוֹצִים לַעֲשׂוֹת לְטוֹבַת בְּרִיאוּתוֹ שֶׁיֵּלְכוּ לְשַׂמֵּחַ יְהוּדִים. אַחַר הַצָּהֳרַיִם נִכְנַס לְחַדְרוֹ הָרַב שְׁלֹמֹה אַהֲרֹן קָזַרְנוֹבְסְקִי )יַחַד עִם הָרַב גְּרוֹנֶר(. כְּשֶׁנִּכְנַס אִיחֵל לְאד"ש רְפוּאָה שְׁלֵמָה וּפָרַץ בִּבְכִי וְאד"ש אָמַר לוֹ: הֲרֵי זוֹ סְתִירָה לְ'וְשָׂמַחְתָּ בְּחַגֶּיךָ', אִם רוֹצִים לְהוֹסִיף בִּבְרִיאוּתִי, הֲרֵי זֶה דַּווְקָא עַל יְדֵי שֶׁיַּרְבּוּ בְּשִׂמְחָה, בְּכִיּוֹת הֵן נֶגֶד רְצוֹנִי. וְנָתַן לוֹ שְׁנֵי בַּקְבּוּקֵי מַשְׁקֶה )שֶׁהִכְנִיסוּ אֵלָיו עוֹד בָּעֶרֶב הֶחָג( וְהוֹרָה שֶׁבַּלַּיְלָה תִּתְקַיֵּים הִתְווַעֲדוּת כָּרָגִיל, שֶׁבְּוַדַּאי לֹא יֶחְסְרוּ בָּהּ דִּבְרֵי תּוֹרָה, הִתְעוֹרְרוּת וְשִׂמְחָה, וְשֶׁאַחֲרֵי יַעַרְכוּ אֶת הַהַקָּפוֹת וְשֶׁיִּהְיוּ בְּשִׂמְחָה וְשֶׁלֹּא יִגָּרַע מִשִּׂמְחַת הֶחָג.
הָרַבִּי מְדַבֵּר עִם הָרוֹפֵא עַל פְּעוּלַּת הַמַּזְרֵק

<span class="num">א.</span> אָסוּר לְבַשֵּׁל לְצוֹרֶךְ גּוֹי בְּיוֹם טוֹב, שֶׁנֶּאֱמַר "אַךְ אֲשֶׁר יֵיאָכֵל לְכָל נֶפֶשׁ הוּא לְבַדּוֹ יֵיעָשֶׂה לָכֶם", לָכֶם וְלֹא לַגּוֹיִים. וְאָסוּר לְבַשֵּׁל גַּם לְצוֹרֶךְ קָרָאִים בְּיוֹם טוֹב. וְכֵן אָסוּר לְבַשֵּׁל לְצוֹרֶךְ יִשְׂרָאֵל שֶׁמְּחַלֵּל שַׁבָּת בְּפַרְהֶסְיָא. לְפִיכָךְ אָסוּר לְהַזְמִינוֹ לִסְעֹוד עִמּוֹ סְעֻודּוֹת יוֹם טוֹב, גְּזֵירָה שֶׁמָּא יְבַשֵּׁל בִּשְׁבִילוֹ ]בִּקְדֵירָה בִּפְנֵי עַצְמָהּ[. וְאִם בָּא גּוֹי אוֹ קָרָאִי אוֹ מְחַלֵּל שַׁבָּת בְּפַרְהֶסְיָא לְבַקְּרוֹ בְּיוֹם טוֹב יֵשׁ אוֹמְרִים שֶׁמֻּתָּר לַעֲשׂוֹת קָפֶה הַרְבֵּה בְּבַת אַחַת, בְּאוֹפֶן שֶׁיִּשְׁתֶּה עִמּוֹ גַּם בַּעַל הַבַּיִת. וְהַנָּכוֹן לְהַחְמִיר, אֲפִילּוּ אִם מַקְדִּים שְׁתִיָּיתָם לִשְׁתִיַּית הַגּוֹי.

<span class="num">ג.</span> עַכּוּ"ם שֶׁהֵבִיא לְיִשְׂרָאֵל בְּיוֹם טוֹב דָּבָר שֶׁיֵּשׁ בְּמִינוֹ בִּמְחֻבָּר, אוֹ מְחוּסַּר צֵידָה, שֶׁיֵּשׁ לַחֲשׁוֹשׁ שֶׁמָּא תְּלָשָׁם בְּיוֹם טוֹב, וְטָעָה הַיִּשְׂרָאֵל וּבֵירַךְ עָלָיו כְּדֵי לְאוֹכְלוֹ, אַף עַל פִּי שֶׁהוּא מֻקְצֶה יֵשׁ לוֹ לִטְעֹום מְעַט מֵהַפְּרִי, כְּדֵי שֶׁלֹּא תִּהְיֶה בְּרָכָתוֹ בְּרָכָה לְבַטָּלָה. וְאִם הָיָה בְּתוֹךְ הַסְּעֻודָּה שֶׁאֵין כָּאן בְּרָכָה, וְטָעָה הַיִּשְׂרָאֵל וּנְתָנוֹ לְתוֹךְ פִּיו וּלְעָסוֹ, אָסוּר לוֹ לְבָלְעוֹ.

<span class="num">א.</span> הֲנָאָה הַשָּׁוָוה לְכָל נֶפֶשׁ מוּתֶּרֶת בְּיוֹם טוֹב כְּאוֹכֶל נֶפֶשׁ עַצְמוֹ, וְלָכֵן מוּתָּר לַעֲשׂוֹת מְדוּרָה לְהִתְחַמֵּם כְּנֶגְדָּהּ, בִּזְמַן הַקֹּר, שֶׁבְּכָל מָקוֹם שֶׁנֶּאֱמַר "יֵיאָכֵל" )הַיּוּ"ד בְּצֵירֵי(, נִכְלָל בָּזֶה גַּם הֲנָאָה, וְהוֹאִיל וְנֶאֱמַר: אֲשֶׁר "יֵיאָכֵל" לְכָל נֶפֶשׁ, הוּתְּרָה הַהֲנָאָה כְּמוֹ הָאֲכִילָה. וּמִטַּעַם זֶה מוּתָּר גַּם כֵּן לְחַמֵּם מַיִם חַמִּים לִרְחֹץ בָּהֶם פָּנָיו יָדָיו וְרַגְלָיו, שֶׁהִיא הֲנָאָה הַשָּׁוָוה "לְכָל נֶפֶשׁ". אֲבָל לִרְחֹץ כָּל גּוּפוֹ אָסוּר, שֶׁאֵין זוֹ הֲנָאָה הַשָּׁוָוה "לְכָל" נֶפֶשׁ. כָּךְ כָּתַב הַטַּעַם בַּחתם"ס שַׁבָּת לט: דְּדָמֵי לְמוּגְמָר, שֶׁמָּא יְחַמֵּם מַיִם, כְּמוֹ בְּשַׁבָּת. וּלְהָרִי"ף וְהָרַמְבַּ"ם וּמָרַן עֶצֶם הַחִמּוּם לְכָל גּוּפוֹ אִיסּוּרוֹ מִדְּרַבָּנָן מִשּׁוּם גְּזֵירַת מֶרְחָץ, וְלָכֵן לֹא אָסְרוּ רְחִיצַת כָּל גּוּפוֹ, מִמַּיִם שֶׁהוּחַמּוּ מֵעֶרֶב יו"ט.
<span class="num">ב.</span> מַיִם שֶׁהוּחַמּוּ בְּעֶרֶב יוֹם טוֹב מוּתָּר לִרְחֹץ בָּהֶם כָּל גּוּפוֹ כְּאֶחָד, שֶׁלֹּא גָּזְרוּ בָּזֶה אֶלָּא בְּשַׁבָּת, וְדַוְקָא חוּץ לַמֶּרְחָץ אֲבָל בְּבֵית הַמֶּרְחָץ אָסוּר. וּבְאַמְבַּטְיָה פְּרָטִית שֶׁנִּמְצֵאת בְּדִירָתוֹ יֵשׁ לְהָקֵל. וְכֵן מַיִם שֶׁהוּחַמּוּ בְּדוּד שֶׁמֶשׁ, מוּתָּר לְהִתְרַחֵץ בָּהֶם כָּל גּוּפוֹ בְּיוֹם טוֹב, כְּשֶׁהוּא בְּאַמְבַּטְיָה פְּרָטִית שֶׁנִּמְצֵאת בְּדִירָתוֹ.
<span class="num">ג.</span> מוּגְמָר אָסוּר בְּיוֹם טוֹב, וְהַיְינוּ לְפַזֵּר אַבְקַת בְּשָׂמִים עַל גַּבֵּי גֶּחָלִים לְהָרִיחַ, אוֹ לְגַמֵּר כֵּלִים, וְכַיּוֹצֵא בָּזֶה. וְהַטַּעַם לְאִיסּוּר זֶה, מִפְּנֵי שֶׁבְּעֵת שֶׁמְּפַזֵּר הַבְּשָׂמִים עַל הַגֶּחָלִים הוּא מְכַבֶּה אוֹתָן וּלְבַסּוֹף מַבְעִירָן, וּמִכֵּיוָן שֶׁאֵין זוֹ הֲנָאָה הַשָּׁוָוה "לְכָל" נֶפֶשׁ, כִּי אִם לָעֲשִׁירִים הַמְּפֻנָּקִים בִּלְבַד אָסוּר. אֲבָל אִם עוֹשֶׂה כְּדֵי לָתֵת רֵיחַ טוֹב בְּפֵירוֹת לְמַתְּקָם לַאֲכִילָה מוּתָּר, וַאֲפִילּוּ אִם הֵם טוֹבִים וּרְאוּיִים לַאֲכִילָה בְּלִי זֶה אֶלָּא שֶׁעַל יְדֵי כָּךְ מַמְתִּיקָם יוֹתֵר מוּתָּר, שֶׁאֵין דִּין מוֹלִיד רֵיחָא בְּאוֹכָלִים. וּמִטַּעַם זֶה מוּתָּר גַּם כֵּן לָתֵת מֵי וְרָדִים בְּכוֹס מַיִם אוֹ יַיִן וּשְׁאָר מַשְׁקִים. וְכֵן מוּתָּר לִשְׁפּוֹךְ טִיפּוֹת מֵי וְרָדִים וְכַדּוֹמֶה לְתוֹךְ טַבָּק הֲרָחָה, שֶׁלְּעִנְיָן זֶה נֶחְשָׁב כְּאוֹכֶל, וְאֵין מוֹלִיד רֵיחָא בְּאוֹכָלִים. אֲבָל אָסוּר לִשְׁפּוֹךְ מֵי וְרָדִים וּשְׁאָר מִינֵי בְּשָׂמִים עַל מִטְפַּחַת כִּיס. וּמִכָּל מָקוֹם מוּתָּר לָתֵת רֵיחַ טוֹב בְּיָדַיִם וּבְפָנִים, שֶׁאֵין זֶה דָּבָר הַמִּתְקַיֵּים, שֶׁזֵּיעַת הָאָדָם מַעֲבִירָתוֹ מֵעַל בְּשָׂרוֹ.
<span class="num">ד.</span> מוּתָּר לְעַשֵּׁן סִיגַרְיוֹת בְּיוֹם טוֹב, וְאַף עַל פִּי שֶׁיֶּשְׁנָם אַחֲרוֹנִים שֶׁאוֹסְרִים, מִפְּנֵי שֶׁמְּדַמִּים אוֹתוֹ לְמוּגְמָר שֶׁאֵינוֹ שָׁוֶוה "לְכָל" נֶפֶשׁ, פָּשַׁט הַמִּנְהָג לְהַתִּיר. וּבִפְרָט בִּזְמַן הָאַחֲרוֹן שֶׁהָעִישּׁוּן נָפוֹץ מְאֹד, וְאִם לֹא יְעַשֵּׁן יִהְיֶה לוֹ צַעַר מֵהַדָּבָר. וּמִכָּל מָקוֹם אָדָם שֶׁאֵינוֹ רָגִיל כָּל כָּךְ בְּעִישּׁוּן וְאֵין לוֹ צַעַר בִּמְנִיעָתוֹ, וּמְכַבְּדִים אוֹתוֹ בְּסִיגַרְיָה בְּיוֹם טוֹב, אוֹ שֶׁרוֹצֶה לְעַשֵּׁן לְשֵׁם תַּעֲנוּג בְּעָלְמָא, נָכוֹן שֶׁיִּמָּנַע מִלְּעַשֵּׁן בְּיוֹם טוֹב. וְתָבוֹא עָלָיו בִּרְכַּת טוֹב. וְכָל זֶה מִצַּד דִּין יוֹם טוֹב, אֲבָל לְגַבֵּי עֶצֶם הָעִישּׁוּן, דַּעַת הָרוֹפְאִים בִּזְמַנֵּנוּ שֶׁהָעִישּׁוּן גּוֹרֵם לְחָלָאִים רַבִּים הַמְּסַכְּנִים אֶת בְּרִיאוּת הָאָדָם, וְשׁוֹמֵר נַפְשׁוֹ יִרְחַק מִזֶּה.
<span class="num">ה.</span> יֵשׁ לְהַתִּיר לְעַשֵּׁן סִיגַרְיוֹת בְּיוֹם טוֹב גַּם כְּשֶׁמֻּדְפָּסוֹת עֲלֵיהֶם אוֹתִיּוֹת שׁוֹנוֹת. וְאַף עַל פִּי שֶׁהוּא מוֹחֵק אֶת הָאוֹתִיּוֹת בְּעֵת שֶׁמְּעַשֵּׁן בִּמְקוֹם הָאוֹתִיּוֹת וְהָאוֹתִיּוֹת נִשְׂרָפוֹת עַל יְדֵי כָּךְ.
<span class="num">ו.</span> יֵשׁ שֶׁנּוֹהֲגִים לְהַחְמִיר שֶׁלֹּא לְהָסִיר אֶת הָאֵפֶר שֶׁנִּשְׂרַף מֵהַסִּיגַרְיָה בְּיוֹם טוֹב, מֵחֲשָׁשׁ שֶׁיִּהְיוּ נִיצוֹצוֹת אֵשׁ נִיתָּזִין עִמּוֹ, וּמִכָּל מָקוֹם אֵין זֶה מֵעִיקַּר הַדִּין.
<span class="num">ז.</span> מִי שֶׁאֵינוֹ מְעַשֵּׁן בְּיוֹם טוֹב מֵחֲמַת שֶׁסָּבוּר שֶׁהַדָּבָר אָסוּר עַל פִּי הַהֲלָכָה, אָסוּר לוֹ גַּם לְטַלְטֵל אֶת הַסִּיגַרְיוֹת כְּדֵי לִיתְּנָם לַחֲבֵירוֹ הַמְּעַשֵּׁן. אוּלָם אִם אֵינוֹ מְעַשֵּׁן בִּגְלַל חֻמְרָא, וְיוֹדֵעַ שֶׁמֵּעִיקַּר הַדִּין מוּתָּר לְעַשֵּׁן בְּיוֹם טוֹב, מוּתָּר לוֹ גַּם לְטַלְטֵל אֶת הַסִּיגַרְיוֹת לִיתְּנָם לַחֲבֵירוֹ.
<span class="num">ח.</span> מוּתָּר לְהַדְלִיק נֵר בְּיוֹם טוֹב כְּדֵי שֶׁיִּהְיֶה מוּכָן לְהִשְׁתַּמֵּשׁ מִמֶּנּוּ בְּיוֹם טוֹב לְבִישּׁוּל וּלְעִישּׁוּן סִיגַרְיָה וְכַדּוֹמֶה.
